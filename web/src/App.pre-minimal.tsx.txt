type Crop = {
  id: string
  name: string
  material: string
  amount: number
  target: number
  percent: number
  complete: boolean
  color: string
}

type Leader = { uuid: string; name: string; total: number; share: number; cropCount: number }
type ActivityItem = { id: number; time: number; player: string; cropId: string; cropName: string; amount: number; type: "pickup" | "reset" }
type HistoryPoint = { time: number; total: number }

type Snapshot = {
  ready: boolean
  generatedAt: number
  challenge: {
    targetPerCrop: number
    cropCount: number
    completedCount: number
    overall: number
    goal: number
    percent: number

    maxPlayers: number

  leaderboard: Leader[]
  activity: ActivityItem[]
  history: HistoryPoint[]

type ConnectionState = "loading" | "live" | "reconnecting"

const number = new Intl.NumberFormat("en", { maximumFractionDigits: 0 })
const compact = new Intl.NumberFormat("en", { notation: "compact", maximumFractionDigits: 1 })

function useDashboard() {
  const [snapshot, setSnapshot] = useState<Snapshot | null>(null)
  const [connection, setConnection] = useState<ConnectionState>("loading")

  useEffect(() => {
    let active = true
    let events: EventSource | null = null

    fetch("/api/v1/progress", { cache: "no-store" })
      .then((response) => {
        if (!response.ok) throw new Error(`HTTP ${response.status}`)
        return response.json() as Promise<Snapshot>
      })
      .then((data) => active && setSnapshot(data))
      .catch(() => active && setConnection("reconnecting"))

    events = new EventSource("/api/v1/events")
    events.addEventListener("snapshot", (event) => {
      if (!active) return
      setSnapshot(JSON.parse((event as MessageEvent).data) as Snapshot)
      setConnection("live")
    })
    events.onerror = () => active && setConnection("reconnecting")

    return () => {
      active = false
      events?.close()
    }
  }, [])

  return { snapshot, connection }
}

function ProgressRing({ percent }: { percent: number }) {
  const radius = 82
  const circumference = 2 * Math.PI * radius
  const offset = circumference * (1 - Math.min(100, Math.max(0, percent)) / 100)
  return (
    <div className="relative grid size-52 shrink-0 place-items-center sm:size-60">
      <div className="absolute inset-5 rounded-full bg-lime-400/[0.04] blur-2xl" />
      <svg className="size-full -rotate-90" viewBox="0 0 200 200" aria-label={`${percent.toFixed(2)} percent complete`}>
        <circle cx="100" cy="100" r={radius} fill="none" stroke="rgba(255,255,255,.06)" strokeWidth="10" />
        <circle
          className="progress-ring"
          cx="100"
          cy="100"
          r={radius}
          fill="none"
          stroke="url(#ring-gradient)"
          strokeLinecap="round"
          strokeWidth="10"
          strokeDasharray={circumference}
          strokeDashoffset={offset}
        />
        <defs>
          <linearGradient id="ring-gradient" x1="0" y1="0" x2="1" y2="1">
            <stop offset="0%" stopColor="#a3e635" />
            <stop offset="60%" stopColor="#84cc16" />
            <stop offset="100%" stopColor="#facc15" />
          </linearGradient>
        </defs>
      </svg>
      <div className="absolute text-center">
        <div className="font-display text-4xl font-bold tracking-[-.06em] text-white sm:text-5xl">{percent.toFixed(2)}<span className="ml-1 text-xl text-lime-300">%</span></div>
        <div className="mt-1 text-[10px] font-bold uppercase tracking-[.24em] text-[#747d71]">Total harvest</div>
      </div>
    </div>
  )
}

function StatCard({ icon: Icon, label, value, note, accent = "lime" }: {
  icon: typeof Trophy
  label: string
  value: string
  note: string
  accent?: "lime" | "gold" | "blue" | "purple"
}) {
  const colors = {
    lime: "bg-lime-400/10 text-lime-300 border-lime-400/10",
    gold: "bg-amber-300/10 text-amber-200 border-amber-300/10",
    blue: "bg-sky-400/10 text-sky-300 border-sky-400/10",
    purple: "bg-fuchsia-400/10 text-fuchsia-300 border-fuchsia-400/10",
  }
  return (
    <Card className="group overflow-hidden">
      <CardContent className="flex items-start justify-between p-5 sm:p-5">
        <div>
          <p className="text-xs font-medium uppercase tracking-[.14em] text-[#747d71]">{label}</p>
          <p className="mt-3 text-2xl font-semibold tracking-[-.04em] text-white">{value}</p>
          <p className="mt-1 text-xs text-[#747d71]">{note}</p>
        </div>
        <div className={cn("rounded-xl border p-2.5 transition-transform group-hover:-rotate-3 group-hover:scale-105", colors[accent])}>
          <Icon className="size-4" />
        </div>
      </CardContent>
    </Card>
  )
}

function ChartTooltip({ active, payload, label, suffix = " crops" }: {
  active?: boolean
  payload?: Array<{ value?: number; color?: string; payload?: { name?: string } }>
  label?: string
  suffix?: string
}) {
  if (!active || !payload?.length) return null
  return (
    <div className="rounded-xl border border-white/10 bg-[#151914]/95 px-3 py-2 shadow-2xl backdrop-blur-xl">
      <p className="text-[10px] font-semibold uppercase tracking-[.15em] text-[#747d71]">{payload[0]?.payload?.name ?? label}</p>
      <p className="mt-1 font-mono text-sm font-semibold text-white">{number.format(payload[0]?.value ?? 0)}{suffix}</p>
    </div>
  )
}

function CropCard({ crop, rank }: { crop: Crop; rank: number }) {
  return (
    <div className="crop-row group rounded-xl border border-white/[0.06] bg-white/[0.018] p-4 transition-colors hover:border-white/[0.12] hover:bg-white/[0.035]">
      <div className="flex items-center gap-3">
        <div className="grid size-10 shrink-0 place-items-center rounded-xl border" style={{ color: crop.color, borderColor: `${crop.color}2c`, backgroundColor: `${crop.color}12` }}>
          {crop.complete ? <Check className="size-4" /> : <Wheat className="size-4" />}
        </div>
        <div className="min-w-0 flex-1">
          <div className="flex items-center justify-between gap-3">
            <div className="flex min-w-0 items-center gap-2">
              <span className="truncate text-sm font-semibold text-[#e9eee5]">{crop.name}</span>
              {crop.complete && <Badge variant="live" className="hidden py-0.5 sm:inline-flex">Done</Badge>}
            </div>
            <span className="font-mono text-xs font-semibold" style={{ color: crop.color }}>{crop.percent.toFixed(1)}%</span>
          </div>
          <div className="mt-2.5 flex items-center gap-3">
            <Progress value={crop.percent} className="h-1" indicatorClassName="crop-progress" />
            <span className="w-20 text-right font-mono text-[10px] text-[#747d71]">{compact.format(crop.amount)} / {compact.format(crop.target)}</span>
          </div>
        </div>
        <span className="hidden w-6 text-right font-mono text-[10px] text-[#4f574d] sm:block">{String(rank).padStart(2, "0")}</span>
      </div>
      <style>{`.crop-row:nth-child(${rank}) .crop-progress { background: ${crop.color}; }`}</style>
    </div>
  )
}

function timeAgo(timestamp: number) {
  const seconds = Math.max(0, Math.floor((Date.now() - timestamp) / 1000))
  if (seconds < 10) return "just now"
  if (seconds < 60) return `${seconds}s ago`
  const minutes = Math.floor(seconds / 60)
  if (minutes < 60) return `${minutes}m ago`
  return `${Math.floor(minutes / 60)}h ago`
}

    <div className="rounded-xl border border-dashed border-white/[0.08] px-5 py-8 text-center">
      <Icon className="mx-auto size-5 text-[#596256]" />
      <p className="mt-3 text-sm font-semibold text-[#a6aea2]">{title}</p>
      <p className="mt-1 text-xs text-[#5e675b]">{copy}</p>
    </div>
  )
}
